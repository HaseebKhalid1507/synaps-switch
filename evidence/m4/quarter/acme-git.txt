+"""Tests for acme-shop#1: empty coupon code crashes checkout with unhandled exception."""
+import pytest
+from shop.catalog import Catalog
+from shop.cart import Cart
+from shop.pricing import apply_coupon
+
+
+# --- apply_coupon directly ---
+
+def test_apply_coupon_empty_string_raises_value_error():
+    """Empty coupon code must raise ValueError, not KeyError."""
+    with pytest.raises(ValueError, match="coupon"):
+        apply_coupon(100.0, "")
+
+
+def test_apply_coupon_unknown_code_raises_value_error():
+    """Unknown coupon code must raise ValueError, not KeyError."""
+    with pytest.raises(ValueError, match="coupon"):
+        apply_coupon(100.0, "DOESNOTEXIST")
+
+
+def test_apply_coupon_valid_code_works():
+    """Valid coupon still applies correctly after the fix."""
+    discounted, discount = apply_coupon(100.0, "SAVE10")
+    assert discount == pytest.approx(10.0)
+    assert discounted == pytest.approx(90.0)
+
+
+# --- via Cart.total() ---
+
+def _make_cart() -> Cart:
+    catalog = Catalog.default()
+    cart = Cart(catalog)
+    # pick the first available product
+    products = catalog.all()
+    assert products, "catalog must have at least one product"
+    cart.add(products[0].id, 1)
+    return cart
+
+
+def test_cart_total_empty_coupon_raises_value_error():
+    """cart.total() with an empty coupon code must raise ValueError, not KeyError."""
+    cart = _make_cart()
+    cart.apply_coupon("")
+    with pytest.raises(ValueError, match="coupon"):
+        cart.total()
+
+
+def test_cart_total_whitespace_coupon_raises_value_error():
+    """Whitespace-only coupon should also be rejected."""
+    cart = _make_cart()
+    cart.apply_coupon("   ")
+    with pytest.raises(ValueError, match="coupon"):
+        cart.total()
+
+
+def test_cart_total_no_coupon_unaffected():
+    """Cart without coupon must still compute total correctly."""
+    cart = _make_cart()
+    result = cart.total()
+    assert result["total"] > 0
+    assert result["discount"] == 0.0
